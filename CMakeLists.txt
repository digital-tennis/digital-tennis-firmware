cmake_minimum_required(VERSION 3.20)
option(DT_ZEPHYR_APP "Build the board application instead of host tests" OFF)
option(DT_IMU_TRACE "Developer UART trace; disabled in ordinary builds" OFF)
if(DT_ZEPHYR_APP)
  set(BOARD_ROOT ${CMAKE_CURRENT_SOURCE_DIR})
  find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
endif()
project(digital_tennis_firmware C)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/config/gesture-rules.csv ${CMAKE_CURRENT_SOURCE_DIR}/config/gesture-detector.csv ${CMAKE_CURRENT_SOURCE_DIR}/tools/generate-gesture-rules.py)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/generate-gesture-rules.py ${CMAKE_CURRENT_SOURCE_DIR}/config/gesture-rules.csv ${CMAKE_CURRENT_BINARY_DIR}/generated/dt_gesture_rules.h ${CMAKE_CURRENT_SOURCE_DIR}/config/gesture-detector.csv RESULT_VARIABLE gesture_result)
if(NOT gesture_result EQUAL 0)
  message(FATAL_ERROR "Invalid gesture rules CSV")
endif()
if(NOT DT_PROTOCOL_ROOT)
  include(FetchContent)
  FetchContent_Declare(protocol GIT_REPOSITORY https://github.com/digital-tennis/digital-tennis-protocol.git GIT_TAG c8a94c4bb62d9c07f293537e087e0e664c306f4d GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(protocol)
else()
  add_subdirectory(${DT_PROTOCOL_ROOT} protocol)
endif()
add_library(dt_controller STATIC src/ap/dt_controller.c src/ap/dt_tx_queue.c src/ap/dt_button.c src/ap/dt_gesture.c)
target_include_directories(dt_controller PUBLIC src/common)
target_include_directories(dt_controller PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(dt_controller PUBLIC dt_protocol)
target_compile_options(dt_controller PRIVATE -Wall -Wextra -Werror)
if(DT_ZEPHYR_APP)
  target_link_libraries(dt_protocol PUBLIC zephyr_interface)
  target_link_libraries(dt_controller PUBLIC zephyr_interface)
  if(BOARD MATCHES "^nu54v_dk(/|$)")
    target_compile_definitions(app PRIVATE DT_BUTTON_PROFILE=1)
  endif()
  target_sources(app PRIVATE src/main.c src/hw/driver/external_imu.c src/hw/driver/buttons.c src/ap/ble_peripheral.c)
  if(DT_IMU_TRACE)
    target_compile_definitions(app PRIVATE DT_IMU_TRACE=1)
    target_sources(app PRIVATE src/hw/driver/imu_trace.c)
  endif()
  target_link_libraries(app PRIVATE dt_controller)
else()
  enable_testing()
  add_executable(controller_checks tests/controller_checks.c)
  target_link_libraries(controller_checks PRIVATE dt_controller m)
  target_compile_options(controller_checks PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME external_imu_pipeline COMMAND controller_checks)
  add_executable(transport_checks tests/transport_checks.c)
  target_link_libraries(transport_checks PRIVATE dt_controller m)
  target_compile_options(transport_checks PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME transport_and_buttons COMMAND transport_checks)
  add_executable(peak_checks tests/peak_checks.c)
  target_link_libraries(peak_checks PRIVATE dt_controller m)
  target_compile_options(peak_checks PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME acceleration_peak_pipeline COMMAND peak_checks)
  add_executable(imu_replay tools/imu_replay.c)
  target_link_libraries(imu_replay PRIVATE dt_controller m)
  target_compile_options(imu_replay PRIVATE -Wall -Wextra -Werror)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME imu_capture_replay COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/replay_checks.py $<TARGET_FILE:imu_replay>)
  add_executable(physical_checks tests/physical_checks.c)
  target_link_libraries(physical_checks PRIVATE dt_controller m)
  target_compile_options(physical_checks PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME physical_event_boundaries COMMAND physical_checks)
  add_executable(gesture_checks tests/gesture_checks.c)
  target_link_libraries(gesture_checks PRIVATE dt_controller m)
  target_compile_options(gesture_checks PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME tennis_gesture_features COMMAND gesture_checks)
  add_test(NAME kinematic_virtual_imu COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/simulator_checks.py $<TARGET_FILE:imu_replay>)
  add_test(NAME threshold_study_boundaries COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/threshold_checks.py $<TARGET_FILE:imu_replay>)
endif()
